Developer Experience

One Query. Every Model.

Mix and match data models in a single transaction. No ETL pipelines, no data duplication, no consistency nightmares.

multi_model_query.sql 
-- Multi-model query: SQL JOIN with vector search
SELECT p.id, p.name, p.price, v.similarity
FROM products p
JOIN vector_search(
  p.embedding,
  query_vector '[0.12, 0.34, ...]',
  top_k 10
) v ON p.id = v.id
WHERE p.category = 'electronics'
  AND p.price < 500
  AND graph_path('purchased_with', p.id) @> '{123, 456}'
ORDER BY v.similarity DESC;

-- Time series aggregation with full-text filtering
SELECT
  time_bucket('1 hour', ts.time) AS hour,
  AVG(ts.temperature) AS avg_temp,
  MAX(ts.pressure) AS max_pressure
FROM sensor_data ts
JOIN devices d ON ts.device_id = d.id
WHERE search_match(d.description, 'industrial HVAC')
  AND ts.time > now() - interval '7 days'
GROUP BY hour
ORDER BY hour;

# Connect to VedaDB
client = vedadb.connect("vedadb://localhost:5432/mydb")

# Hybrid query: vector + SQL + graph in one call
results = client.query("""
  SELECT p.*, v.score
  FROM products p
  JOIN vector_search(p.embedding, $1, 20) v ON p.id = v.id
  WHERE graph_path('similar_to', p.id) @> '{42}'
  ORDER BY v.score DESC
""", embedding_vector)

# Cache the result for 5 minutes
client.cache.setex("recommendations:user:123", 300, results)
import { VedaClient } from '@vedadb/client';

const client = await VedaClient.connect({
  host: 'localhost',
  database: 'mydb',
  models: ['sql', 'vector', 'graph', 'cache']
});

// Unified multi-model transaction
const tx = await client.transaction();
await tx.sql`INSERT INTO orders ...`;
await tx.vector.index('embeddings', newEmbedding);
await tx.graph.link(userId, 'purchased', productId);
await tx.cache.invalidate(`user:${userId}:recommendations`);
await tx.commit();
package main

import (
    "context"
    "github.com/vedadb/vedadb-go/v2"
)

func main() {
    client, _ := vedadb.New("vedadb://localhost:5432")
    
    // Multi-model aggregation pipeline
    result, _ := client.Query(context.Background(), `
        SELECT 
            ts.hour,
            AVG(ts.value) as avg,
            vector_distance(p.embedding, $1) as similarity
        FROM time_series ts
        JOIN products p ON ts.product_id = p.id
        WHERE search_match(p.name, 'laptop*')
        GROUP BY ts.hour
        ORDER BY similarity DESC
        LIMIT 100
    `, embedding)
}
